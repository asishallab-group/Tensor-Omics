\begin{recipe}[label=sub:gene-family-detection]{Gene Family Detection}

\recipepurpose{Group protein-coding genes across species into gene families so
  that downstream analyses can compare orthologous and paralogous copies. This
  is a shared preprocessing step reused by several analysis recipes.}

\nomaintainer

\begin{requiredinputs}
  \item One protein FASTA per species (primary transcripts only).
  \item A species list / directory layout expected by OrthoFinder.
\end{requiredinputs}

\begin{workflow}
  \step{Run OrthoFinder on the per-species protein FASTAs to obtain
        orthogroups.}
  \step{Optionally refine large orthogroups with MCL clustering to split fused
        families.}
  \step{Emit a gene-to-family table used by downstream recipes.}
\end{workflow}

\examplescript{scripts/gene_family_detection.sh}{%
  Wraps OrthoFinder + optional MCL refinement and writes
  \file{orthogroups.tsv}. A default run on the example proteomes reproduces the
  family table used elsewhere in this manual.}

\begin{parameters}
  \param{inflation (-I)}{MCL granularity when refining orthogroups}{%
    Higher (\code{2.0}--\code{4.0}) yields smaller, tighter families; lower
    yields larger, more inclusive families}
\end{parameters}

\begin{pitfall}
  Including multiple splice isoforms per gene inflates family sizes and biases
  every downstream comparison. Reduce each proteome to primary transcripts
  first.
\end{pitfall}

\begin{references}
  \reference{Emms \& Kelly (2019). OrthoFinder: phylogenetic orthology
    inference for comparative genomics. \emph{Genome Biology} 20:238.}
\end{references}

\end{recipe}
