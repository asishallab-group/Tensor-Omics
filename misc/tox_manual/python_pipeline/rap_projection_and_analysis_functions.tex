\subsubsection{Python RAP Projection and Analysis Functions}

The Python interface provides comprehensive wrappers for RAP (Relative Axes Projection) analysis with automatic memory management, input validation, and error handling.

\textbf{Python Implementation Features:}
\begin{itemize}
    \item \textbf{Automatic Memory Layout}: Uses Fortran order for matrices and C order for vectors as required by underlying Fortran code
    \item \textbf{Comprehensive Input Validation}: Validates array dimensions, types, and consistency before Fortran calls
    \item \textbf{Flexible Mask Input}: Accepts both boolean and integer arrays for selection masks
    \item \textbf{Automatic Dimension Handling}: Infers dimensions from input arrays and handles 2D/3D edge cases
    \item \textbf{Read-only Outputs}: All return arrays are marked as read-only to prevent accidental modification
    \item \textbf{Error Conversion}: Converts Fortran error codes to informative Python exceptions
\end{itemize}

\textbf{Validation Checks:}
\begin{itemize}
    \item Vector dimensions must match between inputs
    \item Selection mask lengths must match corresponding array dimensions
    \item For vector field projection, input must have exactly 2*n\_axes rows
    \item For dimensions > 3, selected axes must be valid indices and have exactly 3 elements
    \item All arrays are converted to appropriate data types (float64 for vectors, int32 for masks)
\end{itemize}

\textbf{Usage Example:}
\begin{lstlisting}[language=Python, caption={Complete Python RAP Analysis Workflow}]
import numpy as np

# Example: 4D vectors, 8 vector pairs
n_dims, n_vecs = 4, 8

# Create sample data
origins = np.random.rand(n_dims, n_vecs).astype(np.float64)
targets = np.random.rand(n_dims, n_vecs).astype(np.float64)

# Single vector angle calculation
v1 = origins[:, 0]
v2 = targets[:, 0]
selected_axes = np.array([1, 2, 3], dtype=np.int32)
angle = tox_clock_hand_angle_between_vectors(v1, v2, selected_axes)

# Batch angle calculation
vecs_mask = np.array([True, True, False, True, False, True, True, False])
angles = tox_clock_hand_angles_for_shift_vectors(
    origins, 
    targets, 
    vecs_mask, 
    selected_axes
)

# Axis contribution analysis
shift_vector = v2 - v1
axis_contributions = relative_axes_changes_from_shift_vector(shift_vector)

# RAP projection
expression_matrix = np.random.rand(6, 10).astype(np.float64)
vecs_mask = np.ones(10, dtype=bool)  # Select all vectors
axes_mask = np.array([1, 1, 1, 1, 0, 0], dtype=bool)  # Select first 4 axes
projections = tox_vector_RAP_projection(expression_matrix, vecs_mask, axes_mask)
\end{lstlisting}

\begin{enumerate}
    \item \textbf{Vector RAP Projection}

    Projects selected expression vectors onto the RAP constructed from a selected set of axes.
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{vecs}: 2D numpy array of expression vectors [n\_axes × n\_vecs]
        \item \texttt{vecs\_selection\_mask}: Boolean or integer array of length n\_vecs indicating which vectors to project
        \item \texttt{axes\_selection\_mask}: Boolean or integer array of length n\_axes indicating which axes to include in RAP
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{projections}: 2D numpy array of projected vectors [n\_selected\_axes × n\_selected\_vecs] (read-only)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Vector RAP Projection}]
projections = tox_vector_RAP_projection(vecs, vecs_selection_mask, 
                                        axes_selection_mask)
\end{lstlisting}

    \item \textbf{Field RAP Projection}

    Projects selected vector fields (e.g., shift vectors) onto the RAP constructed from a selected set of axes. Automatically computes shift vectors as differences between origins and targets.
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{vecs}: 2D numpy array of vector fields [2*n\_axes × n\_vecs] (first n\_axes rows: origins, last n\_axes rows: targets)
        \item \texttt{vecs\_selection\_mask}: Boolean or integer array of length n\_vecs indicating which vectors to project
        \item \texttt{axes\_selection\_mask}: Boolean or integer array of length n\_axes indicating which axes to include in RAP
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{projections}: 2D numpy array of projected vectors [n\_selected\_axes × n\_selected\_vecs] (read-only)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Vector Field RAP Projection}]
projections = tox_field_RAP_projection(vecs, vecs_selection_mask, 
                                        axes_selection_mask)
\end{lstlisting}

    \item \textbf{Clock Hand Angle Between Vectors}

    Calculates the signed rotation angle between two RAP-projected and normalized vectors. Handles 2D, 3D, and higher-dimensional cases with automatic directionality calculation.
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{v1}: First normalized vector in RAP space (1D numpy array)
        \item \texttt{v2}: Second normalized vector in RAP space (1D numpy array)
        \item \texttt{selected\_axes\_for\_signed}: Integer array of 3 axes to use for directionality calculation
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{signed\_angle}: Signed angle between vectors in radians [$-\pi$, $\pi$] (read-only float)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Signed Angle Between Vectors}]
angle = tox_clock_hand_angle_between_vectors(v1, v2, 
                                    selected_axes_for_signed)
\end{lstlisting}

    \item \textbf{Clock Hand Angles for Shift Vectors}

    Computes signed rotation angles between corresponding pairs of RAP-projected and normalized vectors (e.g., expression centroids and paralogs).
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{origins}: 2D numpy array of origin vectors [n\_dims × n\_vecs] (e.g., centroids)
        \item \texttt{targets}: 2D numpy array of target vectors [n\_dims × n\_vecs] (e.g., paralogs)
        \item \texttt{vecs\_selection\_mask}: Boolean array of length n\_vecs indicating which vector pairs to compute
        \item \texttt{selected\_axes\_for\_signed}: Integer array of 3 axes to use for directionality calculation
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{signed\_angles}: 1D numpy array of signed angles in radians [$-\pi$, $\pi$] (read-only)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Batch Angle Calculation for Vector Pairs}]
angles = tox_clock_hand_angles_for_shift_vectors(origins, targets, 
                                vecs_selection_mask, selected_axes_for_signed)
\end{lstlisting}

    \item \textbf{Relative Axis Contributions from Shift Vector}

    Computes fractional contribution of each axis to a RAP-projected and normalized shift vector. Values range [0,1] and sum to 1.
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{shift\_vector}: RAP-projected and normalized shift vector (1D numpy array)
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{contrib}: 1D numpy array of relative axis contributions (sums to 1, read-only)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Shift Vector Axis Contributions}]
contributions = relative_axes_changes_from_shift_vector(shift_vector)
\end{lstlisting}

    \item \textbf{Relative Axis Contributions from Expression Vector}

    Computes fractional contribution of each axis to a RAP-projected and normalized expression vector. Values range [0,1] and sum to 1.
    
    \textbf{Arguments:}
    \begin{itemize}
        \item \texttt{expression\_vector}: RAP-projected and normalized expression vector (1D numpy array)
    \end{itemize}
    \textbf{Returns:}
    \begin{itemize}
        \item \texttt{contrib}: 1D numpy array of relative axis contributions (sums to 1, read-only)
    \end{itemize}
\begin{lstlisting}[language=Python, caption={Expression Vector Axis Contributions}]
contributions = relative_axes_expression_from_expression_vector(
    expression_vector)
\end{lstlisting}

\end{enumerate}
